cmake_minimum_required(VERSION 3.16)
if(NOT DEFINED ENV{VITASDK})
  set(ENV{VITASDK} /usr/local/vitasdk)
endif()
set(CMAKE_TOOLCHAIN_FILE "$ENV{VITASDK}/share/vita.toolchain.cmake" CACHE PATH "toolchain")
project(vpk_promoter C)
include("$ENV{VITASDK}/share/vita.cmake" REQUIRED)
add_executable(vpk_promoter main.c)
target_compile_options(vpk_promoter PRIVATE -Wall -Wextra -Werror -std=gnu11)
target_link_libraries(vpk_promoter SceLibKernel_stub SceSysmodule_stub ScePromoterUtil_stub SceIofilemgr_stub)
set(VITA_MAKE_FSELF_FLAGS "-a 0x2808000000000000")
vita_create_self(promoter.bin vpk_promoter UNSAFE)
